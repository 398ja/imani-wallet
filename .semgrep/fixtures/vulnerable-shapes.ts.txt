function idempotencyScope(pubkey: string, key: string): string {
  return `${pubkey.toLowerCase()}\u0000${key}`
}

function check(b: any, defaultNow: number) {
  const now = Number(b.now ?? defaultNow)
  return now
}

function enrolledActor(stored: any) {
  return terminalActor(
    { stallPubkey: stored.stallPubkey, role: stored.role, lockedTo: stored.terminalPubkey },
    stored.terminalPubkey,
  )
}

function leak(signer: any) {

}

function selfCompare(stored: any) {

  return stored
